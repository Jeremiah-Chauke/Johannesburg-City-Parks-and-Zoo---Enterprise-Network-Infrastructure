# Router1 Configuration - IP Telephony
# Device: Tele-Router
# Platform: Cisco 2811
# IOS Version: 15.1
# Project: Johannesburg City Parks and Zoo (JCPZ) Enterprise Network

! ==========================================
! 1. BASIC DEVICE CONFIGURATION
! ==========================================

version 15.1
hostname Tele-Router

no service timestamps log datetime msec
no service timestamps debug datetime msec
no service password-encryption

no ip cef
no ipv6 cef

! ==========================================
! 2. GIGABIT / FAST ETHERNET INTERFACES
! ==========================================

interface FastEthernet0/0
 ip address 192.168.102.66 255.255.255.252
 duplex auto
 speed auto

interface FastEthernet0/1
 ip address 192.168.102.82 255.255.255.252
 duplex auto
 speed auto

! ==========================================
! 3. SERIAL WAN INTERFACES
! ==========================================

interface Serial0/2/0
 ip address 195.136.17.14 255.255.255.252
 clock rate 2000000

interface Serial0/2/1
 ip address 195.136.17.17 255.255.255.252

interface Serial0/3/0
 ip address 195.136.17.2 255.255.255.252

interface Serial0/3/1
 ip address 195.136.17.25 255.255.255.252
 clock rate 64000

! ==========================================
! 4. VLAN INTERFACE
! ==========================================

interface Vlan1
 no ip address
 shutdown

! ==========================================
! 5. STATIC ROUTING
! ==========================================

ip classless
ip route 192.168.100.0 255.255.255.0 192.168.102.81

! ==========================================
! 6. CALL MANAGER EXPRESS CONFIGURATION
! ==========================================

telephony-service
 max-ephones 5
 max-dn 5
 ip source-address 192.168.102.82 port 2000
 auto assign 1 to 5

! ==========================================
! 7. DIRECTORY NUMBERS
! ==========================================

ephone-dn 1
 number 3001

ephone-dn 2
 number 3002

ephone-dn 3
 number 3003

ephone-dn 4
 number 3004

ephone-dn 5
 number 3005
