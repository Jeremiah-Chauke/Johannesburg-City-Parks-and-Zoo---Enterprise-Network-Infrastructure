# CLOUD-SW Configuration
# Device Type: Layer 2 Cloud Environment Switch
# Switching Technology: PVST
# Cloud/WAN VLAN: VLAN 70

enable
configure terminal

! Set device hostname
hostname CLOUD-SW

! Spanning Tree Configuration
spanning-tree mode pvst
spanning-tree extend system-id

! Cloud Environment Access Ports
interface range FastEthernet0/1 - 3
 switchport mode access
 switchport access vlan 70
 no shutdown
exit

! Trunk Link - VLAN 70
interface FastEthernet0/4
 switchport mode trunk
 switchport trunk allowed vlan 70
 no shutdown
exit

! VLAN 70 Interface
interface Vlan70
 no ip address
exit

! Disable VLAN 1 Interface
interface Vlan1
 no ip address
 shutdown
exit

end
write memory
